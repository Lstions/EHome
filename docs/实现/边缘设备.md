```go
type EdgeDevice struct {
    ID              uint           `gorm:"primaryKey" json:"id"`
    Name            string         `gorm:"size:64;not null" json:"name"`
    NodeID          uint           `gorm:"index;not null" json:"node_id"`         // v2.2 新增
    ChannelID       uint           `gorm:"index;not null" json:"channel_id"`     // 改名 (从 device)
    DeviceConfigID  uint           `gorm:"index;not null" json:"device_config_id"` // v2.2 新增 FK
    HardwareID      uint           `gorm:"default:0" json:"hardware_id"`         // v2.2 新增
    IntervalMs      int            `gorm:"default:5000" json:"interval_ms"`
    Enabled         bool           `gorm:"default:true" json:"enabled"`
    Status          string         `gorm:"size:20;default:active" json:"status"`
    ErrorCode       int            `gorm:"default:0" json:"error_code"`           // v2.2 新增
    LastDataAt      *time.Time     `json:"last_data_at"`                          // v2.2 新增
    LastError       string         `gorm:"size:256" json:"last_error"`            // v2.2 新增
    InitState       string         `gorm:"size:20;default:pending" json:"init_state"`  // v2.2 新增
    InitLastStep    int            `gorm:"default:0" json:"init_last_step"`
    InitTotalSteps  int            `gorm:"default:0" json:"init_total_steps"`
    CreatedAt       time.Time      `json:"created_at"`
    UpdatedAt       time.Time      `json:"updated_at"`
    DeletedAt       gorm.DeletedAt `gorm:"index" json:"-"`
    // 关联 (v2.2)
    Node          Node          `gorm:"foreignKey:NodeID" json:"node,omitempty"`
    Channel       Channel       `gorm:"foreignKey:ChannelID" json:"channel,omitempty"`
    DeviceConfig  DeviceConfig  `gorm:"foreignKey:DeviceConfigID" json:"device_config,omitempty"`
}

// v2.1 兼容别名
type Device = EdgeDevice  // v2.1 代码可继续用 Device, 实际指 EdgeDevice
```

**关键 v2.2 变化**:
- 加 `NodeID` FK (v2.1 通过 ChannelID 间接关联, v2.2 显式)
- 加 `DeviceConfigID` FK (关键! v2.1 没有)
- 加 `HardwareID` 字段 (v2.1 在 Channel 端, v2.2 移到 EdgeDevice 端)
- 加 `ErrorCode` / `LastError` / `LastDataAt` 字段 (状态跟踪)
- 加 `InitState` / `InitLastStep` / `InitTotalSteps` 字段 (G6 修复)
- 删 `Type` / `ParserID` (从 device_config 推导, 避免冗余)

## 关键流程 (v2.2 G6 修复)

### 设备初始化 (改用 sendAndWait)

```go
// backend/internal/deviceinit/deviceinit.go

// v2.2: 用 sendAndWait, 不用 time.Sleep
func (o *Orchestrator) InitDevice(deviceID string, channelID uint32, deviceType string) error {
    steps := o.GetInitSequence(deviceType)  // 从 device_config.init_flow 推导
    for i, step := range steps {
        resp, err := o.pendingWrite.SendAndWait(
            deviceID, channelID, step.Data, step.ReadSize, step.TimeoutMs*3)
        if err != nil {
            return fmt.Errorf("step %d %s failed: %w", i, step.Name, err)
        }
        if !resp.Success {
            return fmt.Errorf("step %d %s nack: %s", i, step.Name, resp.ErrorMsg)
        }
        if step.Name == "read_calib" {
            o.saveCalibData(deviceID, deviceType, resp.Data)
        }
    }
    return nil
}
```

### 边缘设备创建 (v2.2 强制 3 要素)

```go
// handler_edge_device.go Create
func (h *EdgeDeviceHandler) Create(c *gin.Context) {
    var req CreateEdgeDeviceRequest
    c.ShouldBindJSON(&req)
    
    // 1. 验证 node / channel / device_config 存在
    node := h.getNode(req.NodeID)
    if node == nil { return 404 }
    channel := h.getChannel(req.ChannelID)
    if channel == nil { return 404 }
    if channel.NodeID != req.NodeID { return 422 } // channel 必须属于该 node
    dc := h.getDeviceConfig(req.DeviceConfigID)
    if dc == nil { return 404 }
    
    // 2. 验证 UNIQUE (node, channel, hardware_id)
    if h.existsByTriple(req.NodeID, req.ChannelID, req.HardwareID) {
        return 409
    }
    
    // 3. 创建
    ed := models.EdgeDevice{
        Name:           req.Name,
        NodeID:         req.NodeID,
        ChannelID:      req.ChannelID,
        DeviceConfigID: req.DeviceConfigID,
        HardwareID:     req.HardwareID,
        IntervalMs:     req.IntervalMs,
        Enabled:        true,
        Status:         "active",
    }
    h.db.Create(&ed)
    
    // 4. 触发 ConfigChange (v2.1 G1 修复)
    h.eventBus.EmitConfigChange(
        collector.CfgChangeEdgeDevice, collector.CfgActionCreate,
        req.NodeID, ed.ID)
    
    c.JSON(201, ed)
}
```

### 周期采集 (按 edge.interval_ms)

```c
// scheduler.c 周期任务
void scheduler_channel_task(void *pvParameters) {
    channel_runtime_t *ch = (channel_runtime_t *)pvParameters;
    
    while (1) {
        for (int i = 0; i < ch->edge_device_count; i++) {
            edge_device_runtime_t *ed = &ch->edge_devices[i];
            if (!ed->enabled) continue;
            
            // 1. 物理读取
            raw_data_t raw = bus_xfer(ch, ed->hardware_id, ed->parser_id);
            
            // 2. 解析
            sensor_data_t sensors = parser_parse(ed->parser_id, raw);
            
            // 3. 上报 (channel_id 字段)
            msg_handler_send_data_report(
                ch->channel_id,   // 中心端按此找 edge_device
                esp_timer_get_time(),
                sequence++,
                raw.bytes, raw.len,
                0,               // error_code
                0                // request_id
            );
            
            vTaskDelay(pdMS_TO_TICKS(ed->interval_ms));
        }
    }
}
```
