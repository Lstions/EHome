### JSON 字段结构

```go
// Connection (设备连接参数)
type ConnectionConfig struct {
    Protocol      string                 `json:"protocol"`        // UART/I2C/SPI/GPIO/ADC
    DefaultParams map[string]interface{} `json:"default_params"`  // 协议默认参数
    Optional      map[string]interface{} `json:"optional,omitempty"`
}
// 例: {"protocol":"SPI", "default_params":{"clock_hz":10000000, "mode":0}}

// Parser (数据解析器)
type ParserConfig struct {
    ID      string                 `json:"id"`     // 注册的 parser ID
    Options map[string]interface{} `json:"options,omitempty"`
}
// 例: {"id":"bosch.bmp280", "options":{"temperature_unit":"celsius"}}

// InitFlow (业务配置流程, 数组)
type InitFlow struct {
    Steps []InitFlowStep `json:"steps"`
}
type InitFlowStep struct {
    Name          string `json:"name"`                     // "soft_reset", "read_chip_id"
    Data          string `json:"data"`                     // hex string "\\xE0"
    ReadSize      int    `json:"read_size"`                // 期望读取字节数
    TimeoutMs     int    `json:"timeout_ms"`               // 超时 (ms)
    ExpectedValue string `json:"expected_value,omitempty"` // v2.2 新增: 期望返回值
    StoreAs       string `json:"store_as,omitempty"`        // v2.2 新增: 命名存储响应
    Description   string `json:"description,omitempty"`    // v2.2 新增: 运维注释
}
// 例: {"steps":[{"name":"soft_reset","data":"\\xE0","read_size":0,"timeout_ms":100},...]}
```

## 关键流程 (v2.2 G1 修复)

### 创建设备配置

```go
// handler_device_config.go Create
func (h *DeviceConfigHandler) Create(c *gin.Context) {
    var req CreateDeviceConfigRequest
    c.ShouldBindJSON(&req)
    
    // 1. 验证 connection
    if !isValidProtocol(req.Connection.Protocol) {
        return 422, "invalid connection.protocol"
    }
    
    // 2. 验证 parser
    if !h.parserRegistry.Exists(req.Parser.ID) {
        return 422, "parser.id not registered"
    }
    
    // 3. 验证 init_flow
    if err := validateInitFlow(req.InitFlow); err != nil {
        return 422, err.Error()
    }
    
    // 4. 验证 (device_type, name) UNIQUE
    if h.existsByDeviceTypeName(req.DeviceType, req.Name) {
        return 409
    }
    
    // 5. 创建
    dc := models.DeviceConfig{
        Name: req.Name,
        DeviceType: req.DeviceType,
        VendorID: req.VendorID,
        DeviceModelID: req.DeviceModelID,
        Connection: JSON(req.Connection),
        Parser: JSON(req.Parser),
        InitFlow: JSON(req.InitFlow),
        IsDefault: req.IsDefault,
        Status: "active",
    }
    h.db.Create(&dc)
    
    // 6. 推 notification 事件
    h.wsHub.BroadcastEvent("notification", gin.H{
        "type": "device_config_created",
        "device_config_id": dc.ID,
    })
    
    c.JSON(201, dc)
}
```

### 修改设备配置 (G1 修复)

```go
func (h *DeviceConfigHandler) Update(c *gin.Context) {
    id := c.Param("id")
    var req UpdateDeviceConfigRequest
    c.ShouldBindJSON(&req)
    
    // 1. 找到设备配置
    dc, _ := h.getDeviceConfig(id)
    
    // 2. 更新字段
    dc.Connection = JSON(req.Connection)
    dc.Parser = JSON(req.Parser)
    dc.InitFlow = JSON(req.InitFlow)
    h.db.Save(&dc)
    
    // 3. 找出引用此 device_config 的所有 edge_devices
    var edgeDevices []models.EdgeDevice
    h.db.Where("device_config_id = ?", id).Find(&edgeDevices)
    
    // 4. EmitConfigChange (G1 修复) - 影响所有引用此的节点
    for _, ed := range edgeDevices {
        h.eventBus.EmitConfigChange(
            collector.CfgChangeDeviceConfig, collector.CfgActionUpdate,
            ed.NodeID, dc.ID)
    }
    
    c.JSON(200, dc)
}
```

### 删除设备配置 (RESTRICT)

```go
func (h *DeviceConfigHandler) Delete(c *gin.Context) {
    id := c.Param("id")
    
    // RESTRICT: 有 edge_devices 引用时阻止
    var count int64
    h.db.Model(&models.EdgeDevice{}).Where("device_config_id = ?", id).Count(&count)
    if count > 0 {
        return 409, "device_config has edge_devices, please migrate first"
    }
    
    // 软删除
    h.db.Delete(&models.DeviceConfig{}, id)
    
    c.JSON(200, gin.H{"deleted": id})
}
```
