## v2.1 → v2.2 关键变更

### 命名 (主要)

| 维度 | v2.1 | v2.2 |
|------|------|------|
| Go struct | `Collector` | `Node` (含 `type Collector = Node` 别名) |
| Go struct | `Device` | `EdgeDevice` (含 `type Device = EdgeDevice` 别名) |
| 表名 | `collectors` | `nodes` (视图 `collectors` 兼容 6 个月) |
| 表名 | `devices` | `edge_devices` (视图 `devices` 兼容 6 个月) |
| API 路径 | `/api/v1/collectors` | `/api/v1/nodes` (双路径, 6 个月) |
| API 路径 | `/api/v1/devices` | `/api/v1/edge-devices` (双路径) |
| WS 事件 | `collector_status` | `node_status` (双发, 6 个月) |
| WS 事件 | `device_status` | `edge_device_status` (双发) |
| 协议字段 1 | `device_id` | `node_id` (v2.2 协议 bump, 值不变) |
| MQTT topic | `devices/{id}/up\|down` | `nodes/{id}/up\|down` (双订阅, 6 个月) |

### 数据库迁移

```sql
-- 创建新表
CREATE TABLE nodes (...);  -- 字段同 collectors
CREATE TABLE edge_devices (...);  -- 加 device_config_id FK

-- 数据迁移
INSERT INTO nodes SELECT * FROM collectors;
INSERT INTO edge_devices SELECT * FROM devices;

-- 修改 channels FK
ALTER TABLE channels RENAME COLUMN collector_id TO node_id;

-- 兼容视图 (6 个月)
CREATE VIEW collectors AS SELECT * FROM nodes;
CREATE VIEW devices AS SELECT * FROM edge_devices;
```

## 协议字段 (v2.2 兼容 v2.1)

### Hello (0x01, Node→Center) — 8 字段

```c
frame_encode_string(&enc, 1, node_id);              // v2.2 字段名 (值同 v2.1 device_id)
frame_encode_string(&enc, 2, firmware_version);
frame_encode_string(&enc, 3, model);
frame_encode_varint(&enc, 4, channel_count);
// v2.1 新增:
frame_encode_varint(&enc, 5, config_mgr_get_epoch());
frame_encode_bool(&enc, 6, config_mgr_has_manifest());
frame_encode_string(&enc, 7, last_manifest);
frame_encode_string(&enc, 8, "2.2");                 // v2.2 协议版本
```

### ConfigManifest (0x04, Center→Node) — v2.2 edge_devices 扩展

```c
frame_encode_string(&enc, 1, manifest_id);
frame_encode_varint(&enc, 2, config_epoch);
// 原有: channels (含 bus_type, bus_config)
for (ch in channels) {
    sub_enc := SubEncoder();
    sub_enc.EncodeVarint(1, ch.id);
    sub_enc.EncodeVarint(2, ch.hardware_id);
    sub_enc.EncodeString(3, ch.bus_type);
    sub_enc.EncodeString(4, ch.bus_config);
    
    // v2.2 新增: edge_devices 子数组
    for (ed in ch.edge_devices) {
        ed_enc := SubEncoder();
        ed_enc.EncodeVarint(1, ed.edge_device_id);  // 中心端 PK, 节点不感知但保留用于追踪
        ed_enc.EncodeVarint(2, ed.hardware_id);
        ed_enc.EncodeString(3, ed.parser_id);
        ed_enc.EncodeVarint(4, ed.interval_ms);
        ed_enc.EncodeBool(5, ed.enabled);
        // init_flow (v2.2 强化)
        if (ed.init_flow) {
            // 序列化为 steps 数组
            for (step in ed.init_flow.steps) {
                step_enc := SubEncoder();
                step_enc.EncodeString(1, step.name);
                step_enc.EncodeString(2, step.data);
                step_enc.EncodeVarint(3, step.read_size);
                step_enc.EncodeVarint(4, step.timeout_ms);
                ed_enc.EncodeSubFrame(6, step_enc.Bytes());
            }
        }
        sub_enc.EncodeSubFrame(7, ed_enc.Bytes());
    }
    
    enc.EncodeSubFrame(3, sub_enc.Bytes());
}
// v2.1 新增:
frame_encode_string(&enc, 5, sync_id);
frame_encode_varint(&enc, 6, sync_reason);
```
